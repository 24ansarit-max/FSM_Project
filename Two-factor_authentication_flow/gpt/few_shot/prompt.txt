Here is an example of a simple FSM in Verilog with a retry-counter and lockout pattern (clean, security-conscious coding style):

module simple_retry_fsm (
    input clk,
    input reset,
    input attempt,      // pulse: user made an attempt
    input correct,       // 1 if the attempt was correct
    output reg locked,
    output reg [1:0] fail_count
);
    localparam TRY = 1'b0, LOCK = 1'b1;
    reg state;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            state <= TRY;
            fail_count <= 0;
        end else begin
            case (state)
                TRY: begin
                    if (attempt) begin
                        if (correct) begin
                            fail_count <= 0;
                        end else if (fail_count == 2'd2) begin
                            state <= LOCK;
                        end else begin
                            fail_count <= fail_count + 1;
                        end
                    end
                end
                LOCK: state <= LOCK;   // no recovery except reset
                default: state <= TRY;
            endcase
        end
    end

    always @(*) begin
        locked = (state == LOCK);
    end
endmodule
