Here is an example of a simple baud-tick generator in SystemVerilog:

module baud_gen #(parameter DIV = 868) (
    input  logic clk, rst,
    output logic tick
);
    logic [$clog2(DIV)-1:0] cnt;
    always_ff @(posedge clk) begin
        if (rst) begin cnt <= '0; tick <= 1'b0; end
        else if (cnt == DIV-1) begin cnt <= '0; tick <= 1'b1; end
        else begin cnt <= cnt + 1'b1; tick <= 1'b0; end
    end
endmodule

Here is an example of a simple serial transmitter FSM:

module ser_tx (
    input  logic       clk, rst, tick, start,
    input  logic [7:0] data,
    output logic       txd, busy
);
    typedef enum logic [1:0] {IDLE, START, DATA, STOP} state_t;
    state_t state, next_state;
    logic [2:0] bit_idx;
    logic [7:0] shreg;

    always_ff @(posedge clk) begin
        if (rst) begin state <= IDLE; bit_idx <= '0; shreg <= '0; end
        else begin
            state <= next_state;
            if (state == IDLE && start) shreg <= data;
            if (state == DATA && tick) begin
                shreg   <= {1'b0, shreg[7:1]};
                bit_idx <= bit_idx + 1'b1;
            end
            if (state != DATA) bit_idx <= '0;
        end
    end

    always_comb begin
        next_state = state;
        txd  = 1'b1;
        busy = (state != IDLE);
        case (state)
            IDLE:  if (start) next_state = START;
            START: begin txd = 1'b0; if (tick) next_state = DATA; end
            DATA:  begin txd = shreg[0]; if (tick && bit_idx == 3'd7) next_state = STOP; end
            STOP:  if (tick) next_state = IDLE;
            default: next_state = IDLE;
        endcase
    end
endmodule
