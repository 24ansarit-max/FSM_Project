Here is an example of a simple FSM in Verilog with clean, safety-conscious coding conventions (synchronous reset, case-based transitions, an explicit safety override check):

module simple_safety_fsm (
    input clk,
    input reset,
    input enable,
    input fault,          // safety input - must override everything
    output reg motor_on
);
    localparam OFF = 1'b0, ON = 1'b1;
    reg state;

    always @(posedge clk or posedge reset) begin
        if (reset)
            state <= OFF;
        else if (fault)
            state <= OFF;            // fault always forces OFF regardless of other logic
        else begin
            case (state)
                OFF: state <= enable ? ON : OFF;
                ON:  state <= enable ? ON : OFF;
                default: state <= OFF;
            endcase
        end
    end

    always @(*) begin
        motor_on = (state == ON) && !fault;   // output also gated by fault as a second safety layer
    end
endmodule
