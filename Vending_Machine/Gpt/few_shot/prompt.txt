Here is an example of a simple FSM in Verilog with clean coding conventions (synchronous reset, case-based transitions, explicit state parameters):

module simple_fsm (
    input clk,
    input reset,
    input trigger,
    output reg busy
);
    localparam IDLE = 2'b00, ACTIVE = 2'b01, DONE = 2'b10;
    reg [1:0] state, next_state;

    always @(posedge clk or posedge reset) begin
        if (reset)
            state <= IDLE;
        else
            state <= next_state;
    end

    always @(*) begin
        case (state)
            IDLE:   next_state = trigger ? ACTIVE : IDLE;
            ACTIVE: next_state = DONE;
            DONE:   next_state = IDLE;
            default: next_state = IDLE;
        endcase
    end

    always @(*) begin
        busy = (state == ACTIVE);
    end
endmodule
