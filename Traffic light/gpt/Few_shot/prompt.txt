Here is an example of a simple 2-state FSM in Verilog, written in a clean behavioral style:

module simple_fsm (
    input clk,
    input reset,
    output reg state_out
);
    reg state;
    parameter A = 1'b0, B = 1'b1;

    always @(posedge clk or posedge reset) begin
        if (reset)
            state <= A;
        else begin
            case (state)
                A: state <= B;
                B: state <= A;
                default: state <= A;
            endcase
        end
    end

    always @(*) begin
        state_out = state;
    end
endmodule
