Here is an example of a simple 3-state traffic light FSM with a timer in SystemVerilog:

module traffic_light (
    input  logic       clk, rst,
    output logic [2:0] light // {R,Y,G}
);
    typedef enum logic [1:0] {RED, GREEN, YELLOW} state_t;
    state_t state, next_state;
    logic [3:0] timer;

    always_ff @(posedge clk) begin
        if (rst) begin state <= RED; timer <= '0; end
        else begin
            state <= next_state;
            timer <= (state != next_state) ? '0 : timer + 1'b1;
        end
    end

    always_comb begin
        next_state = state;
        light = 3'b100;
        case (state)
            RED:    begin light = 3'b100; if (timer == 4'd9) next_state = GREEN;  end
            GREEN:  begin light = 3'b001; if (timer == 4'd9) next_state = YELLOW; end
            YELLOW: begin light = 3'b010; if (timer == 4'd2) next_state = RED;    end
            default: next_state = RED;
        endcase
    end
endmodule

module interlock (
    input  logic req_a, req_b,
    output logic grant_a, grant_b
);
    always_comb begin
        grant_a = req_a & ~req_b;
        grant_b = req_b & ~req_a;
    end
endmodule
