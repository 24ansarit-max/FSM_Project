Here is an example of a simple synchronous memory (single read/write port, same clock) in SystemVerilog:

module sync_mem #(parameter WIDTH = 8, DEPTH = 16) (
    input  logic                     clk,
    input  logic                     we,
    input  logic [$clog2(DEPTH)-1:0] addr,
    input  logic [WIDTH-1:0]         din,
    output logic [WIDTH-1:0]         dout
);
    logic [WIDTH-1:0] mem [0:DEPTH-1];
    always_ff @(posedge clk) begin
        if (we) mem[addr] <= din;
        dout <= mem[addr];
    end
endmodule

Here is an example of an up/down counter used for tracking fill level:

module updown_counter #(parameter WIDTH = 5) (
    input  logic             clk, rst,
    input  logic             inc, dec,
    output logic [WIDTH-1:0] count
);
    always_ff @(posedge clk) begin
        if (rst) count <= '0;
        else if (inc && !dec) count <= count + 1'b1;
        else if (dec && !inc) count <= count - 1'b1;
    end
endmodule
