"Here is an example of a simple traffic light FSM implementation:

module traffic_light (
  input  logic clk, rst,
  output logic [1:0] light // 00=Red, 01=Yellow, 10=Green
);
  typedef enum logic [1:0] {RED, GREEN, YELLOW} state_t;
  state_t state, next_state;
  logic [3:0] timer;

  always_ff @(posedge clk or posedge rst)
    if (rst) begin
      state <= RED;
      timer <= 0;
    end else begin
      state <= next_state;
      timer <= (state != next_state) ? 0 : timer + 1;
    end

  always_comb begin
    next_state = state;
    case (state)
      RED:    if (timer == 9) next_state = GREEN;
      GREEN:  if (timer == 6) next_state = YELLOW;
      YELLOW: if (timer == 2) next_state = RED;
    endcase
  end

  assign light = state[1:0];
endmodule

Here is an example of a simple vending machine FSM that tracks coin inputs and dispenses an item:

module vending_fsm (
  input  logic clk, rst, coin_in,
  output logic dispense
);
  typedef enum logic [1:0] {IDLE, HAVE_COIN, DISPENSE} state_t;
  state_t state, next_state;

  always_ff @(posedge clk or posedge rst)
    if (rst) state <= IDLE;
    else     state <= next_state;

  always_comb begin
    next_state = state;
    dispense = 1'b0;
    case (state)
      IDLE:       if (coin_in) next_state = HAVE_COIN;
      HAVE_COIN:  next_state = DISPENSE;
      DISPENSE:   begin dispense = 1'b1; next_state = IDLE; end
    endcase
  end
endmodule
